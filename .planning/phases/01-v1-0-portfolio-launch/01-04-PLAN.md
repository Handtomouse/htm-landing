    ```tsx
    'use client';
    // app/work/components/SectorFilter.tsx
    // Client Component: filter chip state. Toggles data-sector-active on the grid wrapper
    // and CSS hides non-matching tiles (rule lives in portfolio.css).
    import { useMemo, useState } from 'react';
    import type { SectorFilterProps } from './types';

    export default function SectorFilter({ cases, children }: SectorFilterProps) {
      const sectors = useMemo(() => {
        const set = new Set<string>(['all']);
        for (const c of cases) {
          if (Array.isArray(c.m) && c.m[0]) set.add(String(c.m[0]).toLowerCase());
        }
        return Array.from(set);
      }, [cases]);
      const [active, setActive] = useState<string>('all');
      return (
        <>
          <div className="sector-chips" role="tablist">
            {sectors.map((s) => (
              <button
                key={s}
                type="button"
                role="tab"
                aria-selected={active === s}
                className={`sector-chip ${active === s ? 'is-active' : ''}`}
                onClick={() => setActive(s)}
              >
                {s}
              </button>
            ))}
          </div>
          <div className="gallery-wrap" data-sector-active={active}>
            {children}
          </div>
        </>
      );
    }
    ```

    ```tsx
    import { cases, getCaseBySlug, getPrevCase, getNextCase } from '@/lib/cases';
    import { notFound } from 'next/navigation';
    import CaseDetail from '../components/CaseDetail';

    export const dynamicParams = false;

    export async function generateStaticParams() {
      return cases.map((c) => ({ slug: c.k }));
    }

    export default async function CasePage(
      { params }: { params: Promise<{ slug: string }> }
    ) {
      const { slug } = await params;
      const c = getCaseBySlug(slug);
      if (!c) notFound();
      const prevCase = getPrevCase(slug);
      const nextCase = getNextCase(slug);
      // Pass cases.length as `total` so the case-num header renders "03 / 19" (not the "03 / 03" tautology).
      return (
        <CaseDetail
          case={c}
          total={cases.length}
          prevCase={prevCase}
          nextCase={nextCase}
        />
      );
    }
    ```

    ```tsx
    'use client';
    // app/work/components/CasePlayer.tsx
    // Client Component: controlled image carousel for case detail.
    // Plain <img> per locked default #6.
    import { useState, useEffect, useCallback } from 'react';
    import type { CasePlayerProps } from './types';

    export default function CasePlayer({ images, alt }: CasePlayerProps) {
      const [idx, setIdx] = useState(0);
      const total = images.length;
      const go = useCallback((delta: number) => {
        setIdx((cur) => (cur + delta + total) % total);
      }, [total]);

      useEffect(() => {
        function onKey(e: KeyboardEvent) {
          if (e.key === 'ArrowRight') go(1);
          if (e.key === 'ArrowLeft') go(-1);
        }
        window.addEventListener('keydown', onKey);
        return () => window.removeEventListener('keydown', onKey);
      }, [go]);

      if (total === 0) return null;
      const current = images[idx];

      return (
        <div className="pStage" role="region" aria-label={`${alt} imagery`}>
          <div className="stage">
            <img src={current.src} alt={current.cap || alt} className="stage-img" />
          </div>
          {current.cap && <p className="stage-cap">{current.cap}</p>}
          <div className="stage-controls">
            <button type="button" onClick={() => go(-1)} aria-label="Previous">{'<'}</button>
            <span>{idx + 1} / {total}</span>
            <button type="button" onClick={() => go(1)} aria-label="Next">{'>'}</button>
          </div>
          <div className="stage-strip" role="tablist">
            {images.map((im, i) => (
              <button
                key={im.src}
                type="button"
                role="tab"
                aria-selected={i === idx}
                className={`strip-thumb ${i === idx ? 'is-active' : ''}`}
                onClick={() => setIdx(i)}
              >
                <img src={im.src} alt="" loading="lazy" />
              </button>
            ))}
          </div>
        </div>
      );
    }
    ```
